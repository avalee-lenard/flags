viewof search_fem = Inputs.text({
  label: "🔍 Search Feminine:",
  placeholder: "Search by country, flower, monarchy, republic, or region..."
})

fem_flags = flags
  .filter(d => d.category === "Feminine")
  .sort((a, b) => (b.is_featured - a.is_featured) || a.country_name.localeCompare(b.country_name))
  .filter(d => {
    const q = search_fem.trim().toLowerCase();
    if (!q) return true;
    return (
      d.country_name.toLowerCase().includes(q) ||
      d.government.toLowerCase().includes(q) ||
      d.subregion.toLowerCase().includes(q) ||
      d.description.toLowerCase().includes(q) ||
      d.code.toLowerCase().includes(q)
    );
  })

html`<div class="flag-grid">
  ${fem_flags.map(d => html`<div class="flag-card fem-card">
    <div class="flag-card-img">
      <img src="w2560/${d.file}" loading="lazy" alt="${d.country_name}">
    </div>
    <div class="flag-card-body">
      <div class="flag-card-title">
        <span>${d.is_featured ? "🌸 " : ""}${d.country_name}</span>
        <span class="flag-card-code">${d.code.toUpperCase()}</span>
      </div>
      <div>
        <span class="pill pill-fem-gov">🏛️ ${d.government}</span>
        <span class="pill pill-fem-reg">🌍 ${d.subregion}</span>
      </div>
      <div class="flag-card-desc">${d.description}</div>
    </div>
  </div>`)}
</div>`

continents = ["All Continents", ...new Set(flags.map(d => d.continent)).values()].sort()
gov_types = ["All Government Styles", ...new Set(flags.map(d => d.government)).values()].sort()

viewof filter_continent = Inputs.select(continents, { label: "Continent:", value: "All Continents" })
viewof filter_gov = Inputs.select(gov_types, { label: "Government:", value: "All Government Styles" })
viewof filter_aes = Inputs.radio(["Both (254)", "🌸 Feminine", "⚔️ Masculine"], { label: "Aesthetic:", value: "Both (254)" })

all_filtered = flags.filter(d => {
  if (filter_continent !== "All Continents" && d.continent !== filter_continent) return false;
  if (filter_gov !== "All Government Styles" && d.government !== filter_gov) return false;
  if (filter_aes.includes("Feminine") && d.category !== "Feminine") return false;
  if (filter_aes.includes("Masculine") && d.category !== "Masculine") return false;
  return true;
})

html`<div class="flag-grid">

      </div>
